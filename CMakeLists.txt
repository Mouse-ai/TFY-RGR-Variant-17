cmake_minimum_required(VERSION 3.16)
project(TFY_RGR_Variant17 VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(tfy_lib STATIC src/language.cpp src/automata.cpp src/grammar.cpp src/tasks.cpp)
target_include_directories(tfy_lib PUBLIC include)
if(MSVC)
  target_compile_options(tfy_lib PRIVATE /W4)
else()
  target_compile_options(tfy_lib PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(rgr src/main.cpp)
target_link_libraries(rgr PRIVATE tfy_lib)
add_executable(rgr_tests tests/tests.cpp)
target_link_libraries(rgr_tests PRIVATE tfy_lib)
enable_testing()
add_test(NAME rgr_17_tests COMMAND rgr_tests)
